cmake_minimum_required(VERSION 3.24)
project(CharaShell VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(CHARASHELL_BUILD_APP "Build Qt desktop application" ON)
option(CHARASHELL_WITH_ASR "Enable native Haas-ja ASR through speech.cpp" OFF)
option(CHARASHELL_NATIVE "Enable local llama.cpp and Irodori inference" OFF)
set(NATIVE_INCLUDE_DIR "" CACHE PATH "Pinned llama.cpp and speech.cpp API headers")
set(MINIAUDIO_INCLUDE_DIR "" CACHE PATH "Directory containing miniaudio.h")
if(CHARASHELL_BUILD_APP)
  find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Quick QuickControls2 Network Concurrent)
  find_path(MINIAUDIO_HEADER miniaudio.h HINTS "${MINIAUDIO_INCLUDE_DIR}" REQUIRED)
  qt_standard_project_setup(REQUIRES 6.5)
  if(Qt6_VERSION VERSION_GREATER_EQUAL 6.8)
    qt_policy(SET QTP0004 NEW)
  endif()
  qt_add_executable(charashell src/main.cpp src/NativeRig.cpp src/NativeRigItem.cpp include/NativeRig.hpp include/NativeRigItem.hpp src/miniaudio.c src/AudioEngine.cpp src/Transcriber.cpp
    src/Thinker.cpp src/Speaker.cpp src/ConversationManager.cpp src/System2Client.cpp src/SpeechCoordinator.cpp src/Settings.cpp src/SetupManager.cpp src/StartupManager.cpp
    include/AudioEngine.hpp include/Transcriber.hpp include/Thinker.hpp
    include/Speaker.hpp include/ConversationManager.hpp include/System2Client.hpp include/SpeechCoordinator.hpp include/Settings.hpp include/SetupManager.hpp include/StartupManager.hpp)
  target_include_directories(charashell PRIVATE include "${MINIAUDIO_HEADER}")
  qt_add_resources(charashell voice_design_reference
    PREFIX "/voice-design"
    BASE "${CMAKE_CURRENT_SOURCE_DIR}/resources"
    FILES resources/voice-design-reference.txt)
  if(CHARASHELL_NATIVE)
    target_sources(charashell PRIVATE src/NativeThinker.cpp src/NativeSpeaker.cpp)
    target_include_directories(charashell PRIVATE "${NATIVE_INCLUDE_DIR}")
    target_compile_definitions(charashell PRIVATE CHARASHELL_NATIVE=1)
  endif()
  if(MSVC)
    target_compile_options(charashell PRIVATE /utf-8)
  endif()
  target_link_libraries(charashell PRIVATE Qt6::Core Qt6::Gui Qt6::Quick Qt6::QuickControls2 Qt6::Network Qt6::Concurrent)
  if(WIN32)
    target_link_libraries(charashell PRIVATE ole32)
  endif()
  if(UNIX AND NOT APPLE)
    target_link_libraries(charashell PRIVATE pthread dl m)
  endif()
  if(CHARASHELL_WITH_ASR)
    find_path(SPEECH_INCLUDE speech.h HINTS "${NATIVE_INCLUDE_DIR}" REQUIRED)
    target_include_directories(charashell PRIVATE "${SPEECH_INCLUDE}")
    target_compile_definitions(charashell PRIVATE CHARASHELL_WITH_ASR=1)
  endif()
  foreach(qml_file Main.qml AvatarView.qml LoadingSpinner.qml Theme.qml)
    set_source_files_properties(qml/${qml_file} PROPERTIES QT_RESOURCE_ALIAS ${qml_file})
  endforeach()
  qt_add_qml_module(charashell URI CharaShell VERSION 1.0 QML_FILES qml/Main.qml qml/AvatarView.qml qml/LoadingSpinner.qml qml/Theme.qml)
  qt_add_resources(avatar_resources resources/native-avatar.qrc)
  target_sources(charashell PRIVATE ${avatar_resources})
  add_custom_command(TARGET charashell POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:charashell>/assets/avatar"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "${CMAKE_CURRENT_SOURCE_DIR}/resources/avatar/source.psd"
      "$<TARGET_FILE_DIR:charashell>/assets/avatar/default.psd")
  qt_add_resources(icon_resources resources/icons.qrc)
  target_sources(charashell PRIVATE ${icon_resources})
  qt_add_resources(prompt_resources resources/prompts.qrc)
  target_sources(charashell PRIVATE ${prompt_resources})
  qt_add_resources(thinking_resources resources/thinking.qrc)
  target_sources(charashell PRIVATE ${thinking_resources})
  qt_add_resources(background_resources resources/backgrounds.qrc)
  target_sources(charashell PRIVATE ${background_resources})
  install(TARGETS charashell RUNTIME DESTINATION bin)
  install(FILES LICENSE DESTINATION bin)
  install(FILES resources/avatar/source.psd DESTINATION bin/assets/avatar RENAME default.psd)
  install(FILES resources/native-rig-NOTICE.txt DESTINATION bin)
  install(FILES resources/icons/lucide/LICENSE DESTINATION bin RENAME lucide-LICENSE.txt)
endif()
